	\subsection{Bézout 等式}
	对 $a \neq 0$, $b \neq 0$,则 $d=gcd(a,b)$ 是使下式存在整数解的最小正整数
	$$ax+by=d$$
	若 $(x,y)$ 是一组解,则全部解为
	$$\left(x+\frac{kb}{\gcd(a,b)}, y-\frac{ka}{\gcd(a,b)}\right), \quad k\in\mathbb{Z}$$

\section{毕达哥拉斯三元组}
 毕达哥拉斯三元组可由下式唯一生成
 \[ a=k\cdot (m^{2}-n^{2}),\ \,b=k\cdot (2mn),\ \,c=k\cdot (m^{2}+n^{2}), \]
 其中 $m > n > 0$,$k > 0$,$m \bot n$,且 $m$、$n$ 一奇一偶。

\section{莫比乌斯函数}
\[
	\mu(n) = \begin{cases} 0 & n \textrm{ 不是无平方因子数}\\ 1 & n \textrm{ 有偶数个质因子}\\ -1 & n \textrm{ 有奇数个质因子}\\\end{cases}
\]
  莫比乌斯反演:
  \[ g(n) = \sum_{d|n} f(d) \Leftrightarrow f(n) = \sum_{d|n} \mu(d)g(n/d) \]
  其它有用的形式:

  $ \sum_{d | n} \mu(d) = [ n = 1] $(非常有用)

  $ g(n) = \sum_{n|d} f(d) \Leftrightarrow f(n) = \sum_{n|d} \mu(d/n)g(d)$

 $ g(n) = \sum_{1 \leq m \leq n} f(\left\lfloor\frac{n}{m}\right \rfloor ) \Leftrightarrow f(n) = \sum_{1\leq m\leq n} \mu(m)g(\left\lfloor\frac{n}{m}\right\rfloor)$
